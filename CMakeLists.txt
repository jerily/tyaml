cmake_minimum_required(VERSION 3.22)
project(tyaml VERSION 0.1.0 LANGUAGES C)
option(BUILD_SHARED_LIBS "Build a loadable Tcl extension" ON)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
find_package(TclStub 8.6 REQUIRED)
find_package(libfyaml 1.0 CONFIG REQUIRED)
# Embed the Tcl facade so static applications only need to register Tyaml_Init.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS library/tyaml.tcl)
file(READ library/tyaml.tcl script HEX)
string(REGEX REPLACE "(..)" "0x\\1," script "${script}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/script.h" "static const char tyaml_script[] = {${script}0};\n")
add_library(tyaml src/tyaml.c src/convert.c)
set_target_properties(tyaml PROPERTIES PREFIX "lib" POSITION_INDEPENDENT_CODE ON C_STANDARD 11)
target_include_directories(tyaml PRIVATE "${TCL_INCLUDE_PATH}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(tyaml PRIVATE libfyaml::fyaml)
target_compile_definitions(tyaml PRIVATE PACKAGE_VERSION="${PROJECT_VERSION}")
if(BUILD_SHARED_LIBS)
  if(APPLE)
    set_target_properties(tyaml PROPERTIES INSTALL_RPATH "@loader_path/..")
  elseif(UNIX)
    set_target_properties(tyaml PROPERTIES INSTALL_RPATH "$ORIGIN/..")
  endif()
  target_compile_definitions(tyaml PRIVATE USE_TCL_STUBS)
  target_link_libraries(tyaml PRIVATE "${TCL_STUB_LIBRARY}")
else()
  target_compile_definitions(tyaml PRIVATE STATIC_BUILD)
endif()
set(pkgdir lib/tyaml${PROJECT_VERSION})
install(TARGETS tyaml DESTINATION ${pkgdir})
if(BUILD_SHARED_LIBS)
  configure_file(pkgIndex.tcl.in pkgIndex.tcl @ONLY)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/pkgIndex.tcl" DESTINATION ${pkgdir})
  enable_testing()
  add_test(NAME tyaml COMMAND "${TCL_TCLSH}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/all.tcl" "${CMAKE_CURRENT_BINARY_DIR}")
endif()
